  \foreach \s [count=\i from 0] in {#1}{%
    \draw (\i,-0.5) rectangle ++(1,1);
    \ifnum\s=2 \draw[-{Stealth[length=1.1mm]}] (\i+0.33,-0.35)--(\i+0.33,0.38);
               \draw[-{Stealth[length=1.1mm]}] (\i+0.67,0.35)--(\i+0.67,-0.38);\fi
    \ifnum\s=1 \draw[-{Stealth[length=1.1mm]}] (\i+0.5,-0.35)--(\i+0.5,0.38);\fi
    \ifnum\s=3 \draw[-{Stealth[length=1.1mm]}] (\i+0.5,0.35)--(\i+0.5,-0.38);\fi
  }\end{tikzpicture}}
